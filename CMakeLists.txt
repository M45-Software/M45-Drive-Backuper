cmake_minimum_required(VERSION 3.22)
project(M45DriveBackuper VERSION 0.3.6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets DBus)
add_executable(m45-drive-backuper src/main.cpp src/window.cpp src/exclusions.cpp)
target_link_libraries(m45-drive-backuper PRIVATE Qt6::Widgets Qt6::DBus)
target_compile_definitions(m45-drive-backuper PRIVATE M45_DRIVE_BACKUPER_VERSION="${PROJECT_VERSION}")
target_compile_options(m45-drive-backuper PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS m45-drive-backuper RUNTIME DESTINATION bin)
install(PROGRAMS backend/m45-drive-backuper.py DESTINATION libexec/m45-drive-backuper RENAME backend)
install(FILES backend/recovery.py backend/reporting.py DESTINATION libexec/m45-drive-backuper)
install(FILES packaging/io.github.m45drivebackuper.desktop DESTINATION share/applications)
install(FILES packaging/m45-drive-backuper-autostart.desktop DESTINATION /etc/xdg/autostart RENAME io.github.m45drivebackuper.desktop)
install(FILES packaging/m45-drive-backuper.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES packaging/io.github.m45drivebackuper.policy DESTINATION share/polkit-1/actions)
install(FILES README.md LICENSE DESTINATION share/doc/m45-drive-backuper)
# Documentation captures are opt-in and never installed with the application.
add_executable(capture-screenshots EXCLUDE_FROM_ALL tools/capture-screenshots.cpp src/window.cpp src/exclusions.cpp)
target_include_directories(capture-screenshots PRIVATE src)
target_link_libraries(capture-screenshots PRIVATE Qt6::Widgets Qt6::DBus)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(test-window tests/test_window.cpp src/window.cpp src/exclusions.cpp)
  target_include_directories(test-window PRIVATE src)
  target_link_libraries(test-window PRIVATE Qt6::Widgets Qt6::DBus Qt6::Test)
  add_test(NAME window COMMAND test-window)
  set_tests_properties(window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME backend COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
  set_tests_properties(backend PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
  add_test(NAME backup-harness COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/backup_harness.py --strict --temporary --output ${CMAKE_BINARY_DIR}/backup-lab)
  set_tests_properties(backup-harness PROPERTIES TIMEOUT 180)
  add_test(NAME preview COMMAND m45-drive-backuper --demo --screenshot ${CMAKE_BINARY_DIR}/preview.png)
  set_tests_properties(preview PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME m45-drive-backuper)
set(CPACK_PACKAGE_CONTACT "M45 Drive Backuper contributors")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "KDE tray configuration and monitoring for local Borg backups")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_PACKAGE_SECTION utils)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3, borgbackup (>= 1.2), systemd, polkitd, pkexec, qt6-qpa-plugins")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "qt6-wayland")
set(CPACK_DEBIAN_PACKAGE_SUGGESTS "k4dirstat")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
include(CPack)
